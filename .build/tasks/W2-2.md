```ts
  const send = async (draft: ChatTurnDraft): Promise<boolean> => {
    if (!api) return false;
    const text = draft.text.trim();
    const conversation = active ?? (await startChat(currentProjectId, text));
    if (!conversation) return false;
    const idempotencyKey = crypto.randomUUID();
    const input = chatTurnInput({ conversationId: conversation.id, idempotencyKey, message: text, profile: 'hub' });
    const userRow: ChatMessageRow = { id: idempotencyKey, conversation_id: conversation.id, kind: 'user', body: text, meta: {}, created_at: new Date().toISOString() };
    setMessages((prev) => [...prev, userRow]);
    setConversations((prev) => prev.map((item) => (item.id === conversation.id ? { ...item, running: true } : item)));
    setStreamingText('');
    const controller = new AbortController();
    abortRef.current = controller;
    let accumulated = '';
    let turnId: string | null = null;
    let failed = false;
    try {
      await submitChatTurn(api, input, {
        signal: controller.signal,
        onEvent: (event) => {
          if (event.type === 'turn') {
            turnId = String(event.data.turn_id ?? '');
            activeTurnRef.current = turnId;
          } else if (event.type === 'text') {
            accumulated += String(event.data.content ?? '');
            setStreamingText(accumulated);
          } else if (event.type === 'tool') {
            const tool = parseToolEventPayload(String(event.data.content ?? ''));
            if (tool && turnId) {
              const toolRow: ChatMessageRow = {
                id: `${turnId}:tool:${tool.call_id}`,
                conversation_id: conversation.id,
                kind: 'tool',
                body: String(event.data.content ?? ''),
                meta: { name: tool.name, call_id: tool.call_id, arguments: tool.arguments },
                created_at: new Date().toISOString(),
              };
              setMessages((prev) => [...prev, toolRow]);
            }
          } else if (event.type === 'error') {
            failed = true;
            accumulated = accumulated || String(event.data.content ?? '');
          }
        },
      });
      if (turnId === null) turnId = activeTurnRef.current;
    } catch {
      failed = !controller.signal.aborted; // user-initiated stop is not a failure
    } finally {
      if (activeTurnRef.current === turnId) activeTurnRef.current = null;
      if (abortRef.current === controller) abortRef.current = null;
      setStreamingText(null);
      setConversations((prev) => prev.map((item) => (item.id === conversation.id ? { ...item, running: false } : item)));
    }
    const stopped = controller.signal.aborted;
    const body = accumulated.trim() || (failed ? 'The turn failed before producing a reply.' : stopped ? 'Stopped.' : 'Acknowledged.');
    const agentRow: ChatMessageRow = {
      id: `${turnId ?? idempotencyKey}:reply`,
      conversation_id: conversation.id,
      kind: 'agent',
      body,
      meta: { turn_id: turnId, state: failed ? 'failed' : 'completed' },
      created_at: new Date().toISOString(),
    };
    setMessages((prev) => [...prev, agentRow]);
    void refreshConversations();
    void refreshFrame();
    return !failed;
  };
```
